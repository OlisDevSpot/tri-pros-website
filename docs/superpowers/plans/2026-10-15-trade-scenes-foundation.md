## Global Constraints

- Verification is `pnpm tsc` and `CI=1 pnpm lint` only. **Never `pnpm build`.** `pnpm lint` is `next lint` and skips `scripts/`; lint scripts with `CI=1 pnpm exec eslint <files>`.
- Pure logic is proved by a `tsx` assertion script under `scripts/` (no test runner; adding one is out of scope). Run with `npx tsx scripts/<name>.ts`.
- **The tree is shared with other sessions.** Commit by explicit path (`git add -- <paths>` then `git commit -m … -- <paths>`, or stage then commit without a pathspec after checking `git diff --cached --stat`). Never `git add -A`, `git stash`, `git reset`, `git checkout -- <file>`, or `--amend`. Never `git push`.
- After any scripted edit (`sed`, Python), run `git diff --shortstat -- <file>`: a whole-file diff means line endings flipped (CRLF). Use `sed -i` or Python with `newline=''`.
- No DB writes, not even for screenshots. Live Notion reads (Tasks 2, 13) are read-only through `catalogSource`; nothing writes to Notion.
- One component per file; file-level constants in `constants/`; hooks in `hooks/`; contexts in `contexts/`; named exports only (`next/dynamic` uses `.then(m => m.X)`); types beside their module (`types.ts`).
- Comments say why, never what; no file banners; no citations of plans, specs or tasks in code. A generated file carries one `// GENERATED by … — do not edit by hand.` line (precedent `src/shared/constants/company/service-area-zips.ts`).
- `three`, `@react-three/*` are imported only under `src/shared/components/scene/`, `src/features/trade-scenes/` and (until phase 3 deletes it) `src/features/meeting-visualizations/`. `src/shared/components/scene/` never imports `@/shared/modules/construction`. Task 3 makes ESLint enforce both.
- Code names keep the ROI calculator's words where code moves from there (`cut`, `combineCuts`, `cutFor`, `CUT_MODES`, `repairsPerYear`). Percent figures are percents: `25` means 25 %.
- Vocabulary is `CONTEXT.md` § *Trade scene terms*. Never "experience", "visualization" or "variant" in new code; the look-dev choices are **options**.
- Labs routes (`/dashboard/labs/**`) are super-admin only.
- The lint config runs the React Compiler rules (`react-hooks/immutability`, `refs`, `purity`, `use-memo`): never write to a memoized object (shader uniforms) during render; do it in `useFrame`, an effect, or an event handler. If `react-hooks/use-memo` rejects `useMemo(create, deps)` in `useOwnedGeometry`, write `useMemo(() => create(), deps)`.

`src/shared/components/scene/lib/damp-value.ts`:

```ts
import { SCENE_MOTION } from '@/shared/components/scene/constants/scene-motion'

export type EasedValue = number | readonly number[]

// The step is clamped so a tab returning from the background eases instead of snapping.
function dampNumber(current: number, target: number, deltaSeconds: number): number {
  const step = Math.min(deltaSeconds, SCENE_MOTION.maxFrameSeconds)
  return current + (target - current) * (1 - Math.exp(-SCENE_MOTION.damping * step))
}

export function dampValue<T extends EasedValue>(current: T, target: T, deltaSeconds: number): T {
  if (typeof current === 'number') {
    return dampNumber(current, target as number, deltaSeconds) as T
  }
  const targets = target as readonly number[]
  return current.map((value, index) => dampNumber(value, targets[index], deltaSeconds)) as unknown as T
}

export function isSettled(current: EasedValue, target: EasedValue): boolean {
  if (typeof current === 'number') {
    return Math.abs(current - (target as number)) <= SCENE_MOTION.settled
  }
  const targets = target as readonly number[]
  return current.every((value, index) => Math.abs(value - targets[index]) <= SCENE_MOTION.settled)
}
```

```ts
'use client'

import type { EasedValue } from '@/shared/components/scene/lib/damp-value'

import { useFrame, useThree } from '@react-three/fiber'
import { useEffect, useLayoutEffect, useRef } from 'react'

import { useSceneMotion } from '@/shared/components/scene/contexts/scene-motion-context'
import { dampValue, isSettled } from '@/shared/components/scene/lib/damp-value'

// The scene renders on demand to spare the iPad's battery, so an easing value asks for frames until it
// settles and then stops asking; at rest the canvas draws nothing.
export function useEased<T extends EasedValue>(target: T, apply: (value: T) => void): void {
  const { reducedMotion } = useSceneMotion()
  const invalidate = useThree(state => state.invalidate)
  const current = useRef<T>(target)
  const latestApply = useRef(apply)
  useLayoutEffect(() => {
    latestApply.current = apply
  })

  const targetKey = typeof target === 'number' ? String(target) : target.join(',')
  useEffect(() => {
    invalidate()
  }, [targetKey, invalidate])

  useFrame((state, delta) => {
    const next = reducedMotion ? target : dampValue(current.current, target, delta)
    current.current = next
    latestApply.current(next)
    if (!isSettled(next, target)) {
      state.invalidate()
    }
  })
}
```

```ts
'use client'

import type { BufferGeometry } from 'three'

import { useEffect, useMemo } from 'react'

// Geometry handed to a mesh through `args` is not disposed by R3F, so the owner disposes it or the iPad leaks GPU memory.
export function useOwnedGeometry<T extends BufferGeometry>(create: () => T, deps: readonly unknown[]): T {
  // eslint-disable-next-line react-hooks/exhaustive-deps
  const geometry = useMemo(create, deps)
  useEffect(() => () => geometry.dispose(), [geometry])
  return geometry
}
```

```tsx
'use client'

import type { ReactNode } from 'react'
import type { SceneRenderer } from '@/shared/components/scene/types'

import { Canvas } from '@react-three/fiber'
import { useReducedMotion } from 'motion/react'
import { useMemo, useState } from 'react'

import { SCENE_DEFAULT_CAMERA } from '@/shared/components/scene/constants/scene-camera'
import { SceneMotionContext } from '@/shared/components/scene/contexts/scene-motion-context'
import { createWebGpuRenderer } from '@/shared/components/scene/lib/create-webgpu-renderer'
import { SceneBloom } from '@/shared/components/scene/ui/scene-bloom'
import { SceneLights } from '@/shared/components/scene/ui/scene-lights'

interface SceneShellProps {
  label: string
  fallback: ReactNode
  renderer?: SceneRenderer
  bloom?: boolean
  camera?: { fov: number, position: [number, number, number] }
  children: ReactNode
}

export function SceneShell({ label, fallback, renderer = 'webgl', bloom = false, camera = SCENE_DEFAULT_CAMERA, children }: SceneShellProps) {
  const reducedMotion = useReducedMotion() ?? false
  const [lost, setLost] = useState(false)
  const motion = useMemo(() => ({ reducedMotion }), [reducedMotion])

  // An iPad under memory pressure can drop the context mid-meeting; the fallback carries the story from there.
  if (lost) {
    return fallback
  }

  return (
    <Canvas
      dpr={[1, 2]}
      frameloop="demand"
      camera={{ fov: camera.fov, near: 0.1, far: 120, position: camera.position }}
      gl={renderer === 'webgpu' ? createWebGpuRenderer : { antialias: true }}
      fallback={fallback}
      aria-label={label}
      role="img"
      onCreated={({ gl }) => {
        gl.domElement.addEventListener('webglcontextlost', () => setLost(true), { once: true })
      }}
    >
      <SceneMotionContext value={motion}>
        <SceneLights />
        {children}
        {/* postprocessing has no WebGPU path yet, which is why WebGPU stays a look-dev option. */}
```

```ts
// The CPU twin of the shader's palette lookup, used by DOM legends so a swatch matches the paint exactly.
export function samplePalette(stops: readonly (readonly [number, number, number])[], t: number): [number, number, number] {
  const scaled = Math.min(Math.max(t, 0), 1) * (stops.length - 1)
  const index = Math.min(Math.floor(scaled), stops.length - 2)
  const f = scaled - index
  const a = stops[index]
  const b = stops[index + 1]
  return [a[0] + (b[0] - a[0]) * f, a[1] + (b[1] - a[1]) * f, a[2] + (b[2] - a[2]) * f]
}
```

```tsx
export function ThermalMaterial({ inner, outer, halfThickness, season, style = 'ironbow', axis = 'y', opacity = 1 }: ThermalMaterialProps) {
  const uniforms = useMemo(() => ({
    uStops: { value: THERMAL_PALETTES[style][season].map(hex => new Color(hex)) },
    uInner: { value: inner },
    uOuter: { value: outer },
    uAxis: { value: AXES[axis].clone() },
    uHalfThickness: { value: halfThickness },
    uOpacity: { value: opacity },

  // eslint-disable-next-line react-hooks/exhaustive-deps
  }), [])

    const color = new Color(hex)

    uniforms.uInner.value = nextInner
    uniforms.uOuter.value = nextOuter
    uniforms.uOpacity.value = nextOpacity
  })
  const invalidate = useThree(state => state.invalidate)
  useLayoutEffect(() => {
    uniforms.uAxis.value.copy(AXES[axis])
    uniforms.uHalfThickness.value = halfThickness
    invalidate()

    <shaderMaterial
      uniforms={uniforms}
      vertexShader={THERMAL_VERTEX_SHADER}
      fragmentShader={THERMAL_FRAGMENT_SHADER}
```

```ts
export function sampleProfile(widths: readonly number[], t: number): number {
  if (widths.length === 1) {
    return widths[0]
  }
  const scaled = Math.min(Math.max(t, 0), 1) * (widths.length - 1)
  const index = Math.min(Math.floor(scaled), widths.length - 2)
  return widths[index] + (widths[index + 1] - widths[index]) * (scaled - index)
```

```ts
import { CatmullRomCurve3, TubeGeometry, Vector3 } from 'three'

import { sampleProfile } from '@/shared/components/scene/lib/sample-profile'

const BASE_RADIUS = 1
const RADIAL_SEGMENTS = 12

// A unit tube whose rings are then scaled to the width profile, so one geometry draws the flow narrowing
// through each part instead of stacking separate tubes.
export function buildTaperedTube(points: readonly [number, number, number][], widths: readonly number[], segments = 48): TubeGeometry {
  const curve = new CatmullRomCurve3(points.map(([x, y, z]) => new Vector3(x, y, z)))
  const tube = new TubeGeometry(curve, segments, BASE_RADIUS, RADIAL_SEGMENTS, false)
  const position = tube.getAttribute('position')
  const center = new Vector3()
  const vertex = new Vector3()
  for (let ring = 0; ring <= segments; ring++) {
    const t = ring / segments
    curve.getPointAt(t, center)
    const scale = sampleProfile(widths, t) / 2 / BASE_RADIUS
    for (let side = 0; side <= RADIAL_SEGMENTS; side++) {
      const index = ring * (RADIAL_SEGMENTS + 1) + side
      vertex.fromBufferAttribute(position, index).sub(center).multiplyScalar(scale).add(center)
      position.setXYZ(index, vertex.x, vertex.y, vertex.z)
    }
  }
  position.needsUpdate = true
  tube.computeVertexNormals()
  return tube
}
```

```tsx
  }, [uniforms, style, invalidate])

  // A scrolling ribbon keeps asking for frames while it is on screen; with reduced motion it holds still
  // and shows the flow by width alone.
  useFrame((state, delta) => {
    if (reducedMotion) {
      return
    }
    uniforms.uTime.value += delta * speed
    state.invalidate()
  })

  return (
    <mesh geometry={geometry}>
      <shaderMaterial
        uniforms={uniforms}
        vertexShader={RIBBON_VERTEX_SHADER}
        fragmentShader={RIBBON_FRAGMENT_SHADER}
        transparent
        depthWrite={false}
        blending={AdditiveBlending}
        toneMapped={false}
      />
    </mesh>
  )
}
```

A branch (the cool roof sending part of the flow back to the sky) is a second `FlowRibbon` whose first point is a point on the parent's path; there is no branch prop.

```ts
import { CanvasTexture, RepeatWrapping, SRGBColorSpace } from 'three'

export type NoiseKind = 'grain' | 'grit' | 'fiber'

// Seeded so a texture looks the same on every open; procedural so no image is downloaded.
function seeded(seed: number): () => number {
  let state = seed
  return () => {
    state = (state * 1664525 + 1013904223) % 4294967296
    return state / 4294967296
  }
}

export function createNoiseTexture(kind: NoiseKind, base: string, seed = 7, size = 256): CanvasTexture {
  const canvas = document.createElement('canvas')
  canvas.width = size
  canvas.height = size
  const context = canvas.getContext('2d')!
  context.fillStyle = base
  context.fillRect(0, 0, size, size)
  const random = seeded(seed)
  if (kind === 'grain') {
    for (let line = 0; line < 70; line++) {
      context.strokeStyle = `rgba(60, 35, 10, ${0.05 + random() * 0.12})`
      context.lineWidth = 0.5 + random() * 1.5
      const y = random() * size
      context.beginPath()
      context.moveTo(0, y)
      context.bezierCurveTo(size * 0.3, y + random() * 8 - 4, size * 0.7, y + random() * 8 - 4, size, y + random() * 6 - 3)
      context.stroke()
    }
  }
  else {
    const count = kind === 'grit' ? 9000 : 2500
    for (let dot = 0; dot < count; dot++) {
      const light = random() > 0.5
      context.fillStyle = light ? `rgba(255, 255, 255, ${random() * 0.18})` : `rgba(0, 0, 0, ${random() * 0.25})`
      const radius = kind === 'grit' ? 0.6 + random() : 1 + random() * 3
      context.fillRect(random() * size, random() * size, radius, kind === 'fiber' ? radius * 4 : radius)
    }
  }
  const texture = new CanvasTexture(canvas)
  texture.wrapS = RepeatWrapping
  texture.wrapT = RepeatWrapping
  texture.colorSpace = SRGBColorSpace
  return texture
}
```

```ts
import type { BillCategory } from '@/shared/modules/construction/core/constants/bill-categories'
import type { BillCutAddOnId } from '@/shared/modules/construction/core/constants/bill-cut-add-ons'
import type { TradeSlug } from '@/shared/modules/construction/core/constants/trade-slugs'
import type { TradeBillCuts } from '@/shared/modules/construction/core/schemas/bill-cuts'
import type { CutPart } from '@/shared/modules/construction/core/types'

interface Picks<T extends TradeSlug> {
  trades: readonly T[]
  addOns: readonly BillCutAddOnId[]
}

// Each cut takes its share of what the previous ones left, so the combined cut can never pass 100%.
export function combineCuts<T extends TradeSlug>({ trades, addOns }: Picks<T>, category: BillCategory, cuts: TradeBillCuts): { parts: CutPart<T | BillCutAddOnId>[], combined: number } {
  const parts: CutPart<T | BillCutAddOnId>[] = trades.flatMap((trade) => {
    const percent = cuts[trade]?.base[category]
    return percent ? [{ source: trade, percent }] : []
  })
  for (const trade of trades) {
    for (const addOn of addOns) {
      const percent = cuts[trade]?.addOns?.[addOn]?.[category]
      if (percent) {
        parts.push({ source: addOn, percent })
      }
    }
  }
  const combined = 1 - parts.reduce((kept, part) => kept * (1 - part.percent / 100), 1)
  return { parts, combined }
}
```

`core/lib/cut-for.ts` (moved from `project-remodel-roi.ts`, taking the bill entry instead of the whole form):

```ts
import type { BillCut, BillEntry, CutPart } from '@/shared/modules/construction/core/types'

function positiveOrZero(value: number | null | undefined): number {
  return value != null && Number.isFinite(value) && value > 0 ? value : 0
}

// A typed % or $ replaces the trades' combined cut for that bill; a $ cut never takes the bill below zero.
export function cutFor<S extends string>(entry: BillEntry, parts: CutPart<S>[], combined: number): BillCut<S> {
  const bill = positiveOrZero(entry.now)
  const { mode, value } = entry.cut
  if (mode === 'percent' && value != null) {
    const share = Math.min(100, positiveOrZero(value)) / 100
    return { bill, parts, combined: share, after: bill * (1 - share), typed: true }
  }
  if (mode === 'amount' && value != null) {
    const saved = Math.min(bill, positiveOrZero(value))
    return { bill, parts, combined: bill > 0 ? saved / bill : 0, after: bill - saved, typed: true }
  }
  return { bill, parts, combined, after: bill * (1 - combined), typed: false }
}
```

```ts
  detail: string
```

```ts
import type { Assembly, ExistingConditions, PartCopy, PartDefinition, ResolvedPart, ScopeInstall } from '@/shared/modules/construction/core/types'

interface ResolveInput {
  existing: ExistingConditions
  proposedScopeIds: readonly string[]
  view: 'existing' | 'proposed'
}

const NO_INSTALLS: Partial<Record<string, ScopeInstall>> = {}

function copyFor(part: PartDefinition, material: string | null, condition: ResolvedPart['condition']): PartCopy {
  if (material === null) {
    return part.absentCopy ?? { label: `No ${part.id.replaceAll('-', ' ')}`, detail: '' }
  }
  const definition = part.materials[material]
  return definition.conditionCopy?.[condition] ?? definition.copy
}

// One resolver for every assembly: parts, forms and copy are data, so a new assembly never adds a branch here.
export function resolveAssembly(assembly: Assembly, { existing, proposedScopeIds, view }: ResolveInput, installs: Partial<Record<string, ScopeInstall>> = NO_INSTALLS): ResolvedPart[] {
  const installed = new Map<string, string>()
  if (view === 'proposed') {
    for (const scopeId of proposedScopeIds) {
      const entry = installs[scopeId]
      if (entry?.assembly !== assembly.id) {
        continue
      }
      for (const { part, material } of entry.installs) {
        installed.set(part, material)
      }
    }
  }

  return assembly.parts
    .filter(part => !part.forms || (existing.form !== null && part.forms.includes(existing.form)))
    .map((part) => {
      const installedMaterial = installed.get(part.id)
      const material = installedMaterial ?? existing.parts[part.id]?.material ?? null
      const condition = installedMaterial ? 'installed' : existing.parts[part.id]?.condition ?? 'absent'
      const copy = copyFor(part, material, condition)
      const definition = material === null ? null : part.materials[material]
      return {
        id: part.id,
        group: part.group,
        role: part.role,
        material,
        condition,
        label: copy.label,
        detail: copy.detail,
        thermal: definition?.thermal ?? null,
        water: definition?.water ?? null,
      }
    })
}
```

```ts
import type { FlowProfile, ResolvedPart, Season } from '@/shared/modules/construction/core/types'

import { FLOW_ENDS, SURFACE_FILM_R_VALUE } from '@/shared/modules/construction/core/constants/flow-model'

// Series resistance: a reflective surface first turns back its share of the sun, then each part's R-value
// divides what remains. Summer flows outside → in; winter flows inside → out. Drawing only.
export function relativeHeatFlow(parts: ResolvedPart[], season: Season): FlowProfile {
  const fromSource = season === 'summer' ? [...parts].reverse() : parts
  const present = fromSource.filter(part => part.condition !== 'absent')

  const reflected = season === 'summer'
    ? 1 - present.reduce((kept, part) => kept * (1 - (part.thermal?.reflectance ?? 0)), 1)
    : 0
  const entering = 1 - reflected
  const totalR = SURFACE_FILM_R_VALUE + present.reduce((sum, part) => sum + (part.thermal?.rValue ?? 0), 0)

  const { source, sink } = FLOW_ENDS[season]
  const boundaries: number[] = []
  const temperatures: FlowProfile['parts'] = {}
  let crossedR = 0
  const sourceTemperature = sink + (source - sink) * entering
  const temperatureAt = (r: number) => sourceTemperature - (sourceTemperature - sink) * (r / totalR)

  boundaries.push(entering * SURFACE_FILM_R_VALUE / (SURFACE_FILM_R_VALUE + crossedR))
  for (const part of fromSource) {
    const r = part.condition === 'absent' ? 0 : part.thermal?.rValue ?? 0
    const sourceSide = temperatureAt(crossedR)
    crossedR += r
    const sinkSide = temperatureAt(crossedR)
    // Parts come in inside → out; "inner" is the face toward the living space whatever the season.
    temperatures[part.id] = season === 'summer' ? { outer: sourceSide, inner: sinkSide } : { inner: sourceSide, outer: sinkSide }
    boundaries.push(entering * SURFACE_FILM_R_VALUE / (SURFACE_FILM_R_VALUE + crossedR))
  }

  return { boundaries, parts: temperatures, reflected }
}
```

`core/lib/relative-water-flow.ts`:

```ts
import type { FlowProfile, ResolvedPart } from '@/shared/modules/construction/core/types'

// Water enters only where something is watered: the demand of the surface, and only while irrigation exists to
// supply it. Each part below passes on its share. Drawing only.
export function relativeWaterFlow(parts: ResolvedPart[]): FlowProfile {
  const fromSurface = [...parts].reverse()
  const irrigated = parts.some(part => part.role === 'irrigate' && part.condition !== 'absent')
  const demand = irrigated ? Math.max(0, ...parts.map(part => part.water?.demand ?? 0)) : 0

  const boundaries = [demand]
  const levels: FlowProfile['parts'] = {}
  let flow = demand
  for (const part of fromSurface) {
    const next = part.condition === 'absent' ? flow : flow * (part.water?.passThrough ?? 1)
    levels[part.id] = { outer: flow, inner: next }
    flow = next
    boundaries.push(flow)
  }
  return { boundaries, parts: levels, reflected: 0 }
}
```
